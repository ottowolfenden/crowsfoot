\tikzset{
    cf start/.style={
        decoration={
            markings,
            mark=at position 0 with {
                \draw (0.5,0) -- (0,0.3);
                \draw (0.5,0) -- (0,-0.3);
            }
        },
        postaction={decorate}
    },
    cf end/.style={
        decoration={
            markings,
            mark=at position 1 with {
                \draw (-0.5,0) -- (0,0.3);
                \draw (-0.5,0) -- (0,-0.3);
            }
        },
        postaction={decorate}
    },
    circ start/.style={
        decoration={
            markings,
            mark=at position 0 with {
                \node[circle, draw, minimum size=0.5cm] at (0.75,0) {};
            }
        },
        shorten <=1cm,
        postaction={decorate}
    },
    circ end/.style={
        decoration={
            markings,
            mark=at position 1 with {
                \node[circle, draw, minimum size=0.5cm] at (-0.75,0) {};
            }
        },
        shorten >=1cm,
        postaction={decorate}
    }
}
